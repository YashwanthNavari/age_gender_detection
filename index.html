First-Time Setup: Setting up Python environment and downloading AI models (~100MB). This may take 1-3 minutes.
opencv-python numpy import os import cv2 import numpy as np import gradio as gr from pyodide.http import open_url # 1. Download models using pyodide MODELS_DIR = "models" os.makedirs(MODELS_DIR, exist_ok=True) models = { "age_deploy.prototxt": "https://raw.githubusercontent.com/YashwanthNavari/age_gender_detection/main/models/age_deploy.prototxt", "age_net.caffemodel": "https://raw.githubusercontent.com/YashwanthNavari/age_gender_detection/main/models/age_net.caffemodel", "deploy.prototxt": "https://raw.githubusercontent.com/YashwanthNavari/age_gender_detection/main/models/deploy.prototxt", "gender_deploy.prototxt": "https://raw.githubusercontent.com/YashwanthNavari/age_gender_detection/main/models/gender_deploy.prototxt", "gender_net.caffemodel": "https://raw.githubusercontent.com/YashwanthNavari/age_gender_detection/main/models/gender_net.caffemodel", "res10_300x300_ssd_iter_140000.caffemodel": "https://raw.githubusercontent.com/YashwanthNavari/age_gender_detection/main/models/res10_300x300_ssd_iter_140000.caffemodel" } def load_models_safely(): for name, url in models.items(): path = os.path.join(MODELS_DIR, name) if not os.path.exists(path): print(f"Downloading {name}...") content = open_url(url).read() with open(path, "wb") as f: if isinstance(content, str): f.write(content.encode('utf-8')) else: f.write(content) print(f"Downloaded {name}!") load_models_safely() FACE_PROTO = "models/deploy.prototxt" FACE_MODEL = "models/res10_300x300_ssd_iter_140000.caffemodel" AGE_PROTO = "models/age_deploy.prototxt" AGE_MODEL = "models/age_net.caffemodel" GENDER_PROTO = "models/gender_deploy.prototxt" GENDER_MODEL = "models/gender_net.caffemodel" AGE_BUCKETS = ["(0-2)", "(4-6)", "(8-12)", "(15-20)", "(25-32)", "(38-43)", "(48-53)", "(60-100)"] GENDER_LIST = ["Male", "Female"] try: face_net = cv2.dnn.readNet(FACE_MODEL, FACE_PROTO) age_net = cv2.dnn.readNet(AGE_MODEL, AGE_PROTO) gender_net = cv2.dnn.readNet(GENDER_MODEL, GENDER_PROTO) except Exception as e: print(f"Error loading models: {e}") def detect_age_gender(frame): if frame is None: return frame frame_bgr = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR) h, w = frame_bgr.shape[:2] blob = cv2.dnn.blobFromImage(frame_bgr, 1.0, (300, 300), (104.0, 177.0, 123.0)) face_net.setInput(blob) detections = face_net.forward() for i in range(detections.shape[2]): confidence = detections[0, 0, i, 2] if confidence > 0.7: box = detections[0, 0, i, 3:7] * np.array([w, h, w, h]) x1, y1, x2, y2 = box.astype(int) x1, y1 = max(0, x1), max(0, y1) x2, y2 = min(w, x2), min(h, y2) face = frame_bgr[y1:y2, x1:x2] if face.size == 0: continue face_blob = cv2.dnn.blobFromImage( face, 1.0, (227, 227), (78.4263377603, 87.7689143744, 114.895847746), swapRB=False ) gender_net.setInput(face_blob) gender = GENDER_LIST[gender_net.forward()[0].argmax()] age_net.setInput(face_blob) age = AGE_BUCKETS[age_net.forward()[0].argmax()] label = f"{gender}, Age {age}" cv2.rectangle(frame_bgr, (x1, y1), (x2, y2), (0, 255, 0), 2) # Add custom background to text for better readability (text_width, text_height), baseline = cv2.getTextSize(label, cv2.FONT_HERSHEY_SIMPLEX, 0.8, 2) cv2.rectangle(frame_bgr, (x1, y1 - text_height - 10), (x1 + text_width, y1), (0, 255, 0), cv2.FILLED) cv2.putText(frame_bgr, label, (x1, y1 - 5), cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 0, 0), 2) return cv2.cvtColor(frame_bgr, cv2.BGR2RGB) demo = gr.Interface( fn=detect_age_gender, inputs=gr.Image(sources=["webcam", "upload"], streaming=False), outputs=gr.Image(label="AI Analysis Result"), title="Upload or Capture Photo", description="", allow_flagging="never", theme=gr.themes.Base() ) demo.launch()
